import boto3
import botocore.config
import json
from datetime import datetime


def generate_code_using_bedrock(message: str, language: str) -> str:

    body = {
        "messages": [
            {
                "role": "user",
                "content": [
                    {
                        "text": f"Write {language} code for the following instructions:\n{message}"
                    }
                ]
            }
        ],
        "inferenceConfig": {
            "max_new_tokens": 2048,
            "temperature": 0.1,
            "topP": 0.2
        }
    }

    try:
        bedrock = boto3.client(
            "bedrock-runtime",
            region_name="us-east-1",  # or your Bedrock region
            config=botocore.config.Config(
                read_timeout=300,
                retries={"max_attempts": 3}
            )
        )

        model_id = "amazon.nova-lite-v1:0"

        print(f"MODEL ID = {model_id}")
        print(f"REGION = {bedrock.meta.region_name}")

        response = bedrock.invoke_model(
            modelId=model_id,
            body=json.dumps(body),
            contentType="application/json",
            accept="application/json"
        )

        response_content = response["body"].read().decode("utf-8")
        response_data = json.loads(response_content)

        print(json.dumps(response_data, indent=2))

        code = response_data["output"]["message"]["content"][0]["text"]

        return code

    except Exception as e:
        print(f"Error generating the code: {e}")
        return ""


def save_code_to_s3_bucket(code, s3_bucket, s3_key):

    s3 = boto3.client("s3")

    try:
        s3.put_object(
            Bucket=s3_bucket,
            Key=s3_key,
            Body=code
        )
        print("Code saved to S3")

    except Exception as e:
        print(f"Error saving code to S3: {e}")


def lambda_handler(event, context):

    event = json.loads(event["body"])

    message = event["message"]
    language = event["key"]

    print(message, language)

    generated_code = generate_code_using_bedrock(
        message,
        language
    )

    if generated_code:

        current_time = datetime.now().strftime("%H%M%S")

        s3_key = f"code-output/{current_time}.py"
        s3_bucket = "mhnskta"

        save_code_to_s3_bucket(
            generated_code,
            s3_bucket,
            s3_key
        )

    else:
        print("No code was generated")

    return {
        "statusCode": 200,
        "body": json.dumps(
            {
                "message": "Code generation completed",
                "generated": bool(generated_code)
            }
        )
    }
