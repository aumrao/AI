import json
import boto3
import botocore.config
import base64
import traceback
from datetime import datetime


def generate_image(prompt):

    body = {
        "prompt": prompt,
        "output_format": "png"
    }

    try:

        bedrock = boto3.client(
            "bedrock-runtime",
            region_name="us-west-2",
            config=botocore.config.Config(
                read_timeout=300,
                retries={"max_attempts": 3}
            )
        )

        model_id = "stability.stable-image-core-v1:1"

        print(f"MODEL ID = {model_id}")
        print(f"PROMPT = {prompt}")

        print("Invoking Bedrock...")

        response = bedrock.invoke_model(
            modelId=model_id,
            body=json.dumps(body),
            contentType="application/json",
            accept="application/json"
        )

        print("Bedrock response received")

        response_body = json.loads(
            response["body"].read().decode("utf-8")
        )

        print(
            "Bedrock Response:",
            json.dumps(response_body, indent=2)
        )

        if (
            "images" in response_body and
            len(response_body["images"]) > 0
        ):
            return response_body["images"][0]

        raise Exception(
            f"No image found in response: {response_body}"
        )

    except Exception as e:

        print(
            f"Image generation error: {str(e)}"
        )

        print(
            traceback.format_exc()
        )

        return None


def save_image_to_s3(
    image_base64,
    bucket_name,
    s3_key
):

    try:

        s3 = boto3.client("s3")

        image_bytes = base64.b64decode(
            image_base64
        )

        s3.put_object(
            Bucket=bucket_name,
            Key=s3_key,
            Body=image_bytes,
            ContentType="image/png"
        )

        print(
            f"Image saved to s3://{bucket_name}/{s3_key}"
        )

        return True

    except Exception as e:

        print(
            f"S3 upload error: {str(e)}"
        )

        print(
            traceback.format_exc()
        )

        return False


def lambda_handler(event, context):

    try:

        print(
            "Received event:"
        )

        print(
            json.dumps(
                event,
                indent=2
            )
        )

        body_string = event.get(
            "body",
            "{}"
        )

        if event.get(
            "isBase64Encoded",
            False
        ):
            body_string = (
                base64.b64decode(
                    body_string
                )
                .decode("utf-8")
            )

        body = json.loads(
            body_string
        )

        prompt = body.get(
            "message"
        )

        if not prompt:

            return {
                "statusCode": 400,
                "body": json.dumps(
                    {
                        "message":
                        "Prompt is required"
                    }
                )
            }

        image_base64 = generate_image(
            prompt
        )

        if not image_base64:

            return {
                "statusCode": 500,
                "body": json.dumps(
                    {
                        "message":
                        "Failed to generate image"
                    }
                )
            }

        current_time = (
            datetime.now()
            .strftime(
                "%Y%m%d-%H%M%S"
            )
        )

        bucket_name = "mhnskta"

        s3_key = (
            f"output-images/"
            f"image-{current_time}.png"
        )

        upload_success = (
            save_image_to_s3(
                image_base64,
                bucket_name,
                s3_key
            )
        )

        if not upload_success:

            return {
                "statusCode": 500,
                "body": json.dumps(
                    {
                        "message":
                        "Failed to save image to S3"
                    }
                )
            }

        return {
            "statusCode": 200,
            "body": json.dumps(
                {
                    "message":
                    "Image generated successfully",
                    "generated": True,
                    "bucket": bucket_name,
                    "s3_key": s3_key
                }
            )
        }

    except Exception as e:

        print(
            f"Lambda execution error: {str(e)}"
        )

        print(
            traceback.format_exc()
        )

        return {
            "statusCode": 500,
            "body": json.dumps(
                {
                    "message": str(e)
                }
            )
        }
