import boto3
import botocore.config
import json
import base64
from datetime import datetime
from email import message_from_bytes


def extract_text_from_multipart(data):
    msg = message_from_bytes(data)

    text_content = ""

    if msg.is_multipart():
        for part in msg.walk():
            if part.get_content_type() == "text/plain":
                payload = part.get_payload(decode=True)

                if payload:
                    text_content += (
                        payload.decode(
                            "utf-8",
                            errors="ignore"
                        )
                        + "\n"
                    )

    else:
        if msg.get_content_type() == "text/plain":
            payload = msg.get_payload(decode=True)

            if payload:
                text_content += (
                    payload.decode(
                        "utf-8",
                        errors="ignore"
                    )
                    + "\n"
                )

    return text_content.strip() if text_content else None


def generate_summary_from_bedrock(content: str) -> str:

    body = {
        "messages": [
            {
                "role": "user",
                "content": [
                    {
                        "text": f"""
Summarize the following meeting notes.

Meeting Notes:
{content}

Provide the output in the following format:

1. Executive Summary
2. Key Decisions
3. Action Items
4. Next Steps
"""
                    }
                ]
            }
        ],
        "inferenceConfig": {
            "max_new_tokens": 1000,
            "temperature": 0.1,
            "topP": 0.2
        }
    }

    try:

        bedrock = boto3.client(
            "bedrock-runtime",
            region_name="us-east-1",
            config=botocore.config.Config(
                read_timeout=300,
                retries={"max_attempts": 3}
            )
        )

        model_id = "amazon.nova-lite-v1:0"

        print(f"MODEL ID = {model_id}")
        print(f"REGION = {bedrock.meta.region_name}")

        response = bedrock.invoke_model(
            modelId=model_id,
            body=json.dumps(body),
            contentType="application/json",
            accept="application/json"
        )

        response_content = (
            response["body"]
            .read()
            .decode("utf-8")
        )

        response_data = json.loads(response_content)

        print("Bedrock Response:")
        print(json.dumps(response_data, indent=2))

        summary = (
            response_data["output"]
            ["message"]
            ["content"][0]
            ["text"]
        )

        return summary.strip()

    except Exception as e:
        print(f"Error generating summary: {e}")
        return ""


def save_summary_to_s3_bucket(
    summary,
    s3_bucket,
    s3_key
):

    s3 = boto3.client("s3")

    try:

        s3.put_object(
            Bucket=s3_bucket,
            Key=s3_key,
            Body=summary
        )

        print(
            f"Summary saved to "
            f"s3://{s3_bucket}/{s3_key}"
        )

    except Exception as e:
        print(
            f"Error saving summary to S3: {e}"
        )


def lambda_handler(event, context):

    try:

        print(
            "Received event:",
            json.dumps(event)
        )

        # Decode raw MIME email body
        decoded_body = base64.b64decode(
            event["body"]
        )

        # Extract plain text email content
        text_content = (
            extract_text_from_multipart(
                decoded_body
            )
        )

        if not text_content:

            print(
                "No text content found "
                "in email"
            )

            return {
                "statusCode": 400,
                "body": json.dumps(
                    {
                        "message":
                        "Failed to extract email content"
                    }
                )
            }

        print(
            "Successfully extracted email content"
        )

        # Generate summary
        summary = (
            generate_summary_from_bedrock(
                text_content
            )
        )

        if not summary:

            return {
                "statusCode": 500,
                "body": json.dumps(
                    {
                        "message":
                        "Failed to generate summary"
                    }
                )
            }

        # Save summary to S3
        current_time = (
            datetime.now()
            .strftime("%Y%m%d-%H%M%S")
        )

        s3_bucket = "mhnskta"

        s3_key = (
            f"summary-output/"
            f"summary-{current_time}.txt"
        )

        save_summary_to_s3_bucket(
            summary,
            s3_bucket,
            s3_key
        )

        return {
            "statusCode": 200,
            "body": json.dumps(
                {
                    "message":
                    "Summary generated successfully",
                    "s3_key": s3_key,
                    "generated": True
                }
            )
        }

    except Exception as e:

        print(
            f"Lambda execution error: {e}"
        )

        return {
            "statusCode": 500,
            "body": json.dumps(
                {
                    "message": str(e)
                }
            )
        }
